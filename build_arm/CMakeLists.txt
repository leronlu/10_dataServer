cmake_minimum_required(VERSION 3.10)
project(data_server LANGUAGES CXX C)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_BUILD_TYPE DEBUG)
set(CMAKE_CXX_FLAGS "-Wall -g -fpermissive")
add_subdirectory(/home/lw/WORK_Claude/10_lib ${CMAKE_BINARY_DIR}/10_lib)
find_package(PkgConfig REQUIRED)
find_library(LC log4cpp REQUIRED)
find_library(SQ sqlite3 REQUIRED)
find_library(ZM zmq REQUIRED)
find_library(JP jpeg REQUIRED)
find_library(X2 xml2 REQUIRED)
find_library(IC iconv REQUIRED)
find_library(GD gdal)
find_library(XL xlsreader)
set(DAT "${CMAKE_CURRENT_SOURCE_DIR}/..")
include_directories(${DAT}/ ${DAT}/mMsgProtocol ${DAT}/mModBus ${DAT}/mMinmea ${DAT}/mFileManage ${DAT}/mVideoFusion ${DAT}/mCommon ${DAT}/lib /usr/local/include/sqlite3 /home/lw/WORK_Claude/10_lib/ext/zmq /home/lw/WORK_Claude/10_lib/ext/msg_frame /home/lw/WORK_Claude/10_lib/ext/uart_manage /home/lw/WORK_Claude/10_lib/ext/serial_module)
file(GLOB S_MSG "${DAT}/mMsgProtocol/*.cpp")
file(GLOB S_MODBUS "${DAT}/mModBus/*.cpp")
file(GLOB S_MINMEA "${DAT}/mMinmea/*.cpp" "${DAT}/mMinmea/*.c")
file(GLOB S_FILEMANAGE "${DAT}/mFileManage/*.cpp")
file(GLOB S_VIDEOFUSION "${DAT}/mVideoFusion/*.cpp")
file(GLOB S_COMMON "${DAT}/mCommon/*.cpp")
list(FILTER S_COMMON EXCLUDE REGEX ".*VideoParam\\.cpp$")
file(GLOB S_LIB "${DAT}/lib/*.cpp")
set(SOURCES ${DAT}/main.cpp ${DAT}/DownSideDataModule.cpp ${DAT}/SysConfig.cpp /home/lw/WORK_Claude/10_lib/ext/msg_frame/msg_frame.cpp /home/lw/WORK_Claude/10_lib/ext/zmq/direct_zmq_msg_manage.cpp /home/lw/WORK_Claude/10_lib/ext/zmq/zmq_msg_manage_base.cpp ${S_MSG} ${S_MODBUS} ${S_MINMEA} ${S_FILEMANAGE} ${S_VIDEOFUSION} ${S_COMMON} ${S_LIB})
add_executable(data_server ${SOURCES})
set(O ${GD} ${XL})
foreach(l ${O})
    if(NOT "${l}" MATCHES "-NOTFOUND$")
        list(APPEND OF ${l})
    endif()
endforeach()
target_link_libraries(data_server PRIVATE ${LC} ${SQ} ${ZM} ${JP} ${X2} ${IC} log_level crcmake datemanage mutexmanage processmanage timermanage CharSetConv Sqlite3DB STLPackage XmlParser logmanage zmq_component iec_message uartmanage SerialModule ${OF} pthread dl z m)
